import osimport jsonimport timeimport httpxfrom datetime import datetime, timedeltafrom typing import List, Literal, Optional, Dict, Any
from fastapi import FastAPI, Request, HTTPException, Depends, status, APIRouterfrom fastapi.security import OAuth2PasswordBearer, OAuth2PasswordRequestFormfrom pydantic import BaseModel, Fieldfrom openai import OpenAIimport jwtfrom passlib.context import CryptContext
# ---------------------------------------------------------------------------# Global Configurations & Clients# ---------------------------------------------------------------------------app = FastAPI(title="Milo - Autonomous Chief of Staff & Admin Engine")openai_client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
BOT_TOKEN = os.getenv("TELEGRAM_BOT_TOKEN")OPERATOR_CHAT_ID = os.getenv("TELEGRAM_CHAT_ID")TELEGRAM_API = f"https://api.telegram.org/bot{BOT_TOKEN}"
SECRET_KEY = os.getenv("ADMIN_SECRET_KEY", "milo-super-secret-key-change-in-prod")ALGORITHM = "HS256"ACCESS_TOKEN_EXPIRE_MINUTES = 60 * 24
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/admin/login")
# In-memory stores (replace with PostgreSQL / Redis in production)PENDING_ACTIONS: Dict[str, Dict[str, Any]] = {}ADMIN_USERS_DB: Dict[str, Dict[str, Any]] = {}
# ---------------------------------------------------------------------------# Data Models / Schemas# ---------------------------------------------------------------------------class BusinessProfile(BaseModel):    user_id: str    owner_name: str    industry: str    qualifying_criteria: List[str]    disqualifiers: List[str]    calendar_rules: str    brand_voice: str
class LeadQualificationResult(BaseModel):    is_qualified: bool    lead_score: int = Field(ge=0, le=100)    summary: str    recommended_action: Literal["book_meeting", "send_rejection", "request_more_info"]    suggested_reply: str    proposed_minutes_from_now: Optional[int] = Field(        default=1440,         description="Suggested meeting offset in minutes (e.g. 1440 for tomorrow)"    )
class SocialDraftResult(BaseModel):    platform: Literal["linkedin", "twitter_x", "instagram"]    post_copy: str    hook: str    hashtags: List[str]
class TriageIntent(BaseModel):    intent: Literal["lead_qualification", "social_media_draft", "general_inquiry"]    confidence: float
class InboundPayload(BaseModel):    profile: BusinessProfile    sender_contact: str    message: str
class AdminRegisterRequest(BaseModel):    username: str    password: str
# ---------------------------------------------------------------------------# Security & Admin Helpers# ---------------------------------------------------------------------------def verify_password(plain_password: str, hashed_password: str) -> bool:    return pwd_context.verify(plain_password, hashed_password)
def get_password_hash(password: str) -> str:    return pwd_context.hash(password)
def create_access_token(data: dict, expires_delta: Optional[int] = None) -> str:    to_encode = data.copy()    expire = time.time() + (expires_delta or (ACCESS_TOKEN_EXPIRE_MINUTES * 60))    to_encode.update({"exp": expire})    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
def get_current_admin(token: str = Depends(oauth2_scheme)):    credentials_exception = HTTPException(        status_code=status.HTTP_401_UNAUTHORIZED,        detail="Could not validate admin credentials",        headers={"WWW-Authenticate": "Bearer"},    )    try:        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])        username: str = payload.get("sub")        role: str = payload.get("role")        if username is None or role != "admin":            raise credentials_exception    except jwt.PyJWTError:        raise credentials_exception            user = ADMIN_USERS_DB.get(username)    if user is None:        raise credentials_exception    return user
# ---------------------------------------------------------------------------# Calendar & Telegram Helpers# ---------------------------------------------------------------------------def schedule_calendar_event(summary: str, description: str, start_dt: datetime, duration_minutes: int = 30):    end_dt = start_dt + timedelta(minutes=duration_minutes)        if os.path.exists("token.json"):        from google.oauth2.credentials import Credentials        from googleapiclient.discovery import build                creds = Credentials.from_authorized_user_file("token.json", ["https://www.googleapis.com/auth/calendar.events"])        service = build("calendar", "v3", credentials=creds)        event_body = {            "summary": summary,            "description": description,            "start": {"dateTime": start_dt.isoformat(), "timeZone": "UTC"},            "end": {"dateTime": end_dt.isoformat(), "timeZone": "UTC"},        }        created = service.events().insert(calendarId="primary", body=event_body).execute()        return created.get("htmlLink")    else:        # Fallback Google Calendar manual creation link        return f"https://calendar.google.com/calendar/r/eventedit?text={summary}&dates={start_dt.strftime('%Y%m%dT%H%M%SZ')}/{end_dt.strftime('%Y%m%dT%H%M%SZ')}"
async def send_telegram_approval(action_id: str, title: str, details: str, preview: str):    message_text = (        f" *Milo Action Required: {title}*\n\n"        f" *Summary:* {details}\n\n"        f" *Proposed Action / Draft:*\n_{preview}_\n\n"        f"Would you like Milo to proceed?"    )    payload = {        "chat_id": OPERATOR_CHAT_ID,        "text": message_text,        "parse_mode": "Markdown",        "reply_markup": {            "inline_keyboard": [                [                    {"text": " Approve & Book", "callback_data": f"approve:{action_id}"},                    {"text": " Reject / Cancel", "callback_data": f"reject:{action_id}"}                ]            ]        }    }    if BOT_TOKEN and OPERATOR_CHAT_ID:        async with httpx.AsyncClient() as client:            await client.post(f"{TELEGRAM_API}/sendMessage", json=payload)
async def update_telegram_message(chat_id: int, message_id: int, resolved_text: str):    payload = {        "chat_id": chat_id,        "message_id": message_id,        "text": resolved_text,        "parse_mode": "Markdown"    }    if BOT_TOKEN:        async with httpx.AsyncClient() as client:            await client.post(f"{TELEGRAM_API}/editMessageText", json=payload)
# ---------------------------------------------------------------------------# Admin Endpoints Router# ---------------------------------------------------------------------------admin_router = APIRouter(prefix="/admin", tags=["Admin Management"])
@admin_router.post("/setup-initial-admin")def setup_initial_admin(req: AdminRegisterRequest):    if ADMIN_USERS_DB:        raise HTTPException(status_code=400, detail="Admin already registered. Use login endpoint.")        ADMIN_USERS_DB[req.username] = {        "username": req.username,        "hashed_password": get_password_hash(req.password),        "role": "admin"    }    return {"status": "success", "message": f"Admin account '{req.username}' created successfully."}
@admin_router.post("/login")def admin_login(form_data: OAuth2PasswordRequestForm = Depends()):    user = ADMIN_USERS_DB.get(form_data.username)    if not user or not verify_password(form_data.password, user["hashed_password"]):        raise HTTPException(            status_code=status.HTTP_401_UNAUTHORIZED,            detail="Incorrect username or password",            headers={"WWW-Authenticate": "Bearer"},        )        token = create_access_token(data={"sub": user["username"], "role": user["role"]})    return {"access_token": token, "token_type": "bearer"}
@admin_router.get("/me")def get_admin_profile(admin=Depends(get_current_admin)):    return {"username": admin["username"], "role": admin["role"], "access": "full_control"}
@admin_router.get("/queue")def view_all_pending_approvals(current_admin=Depends(get_current_admin)):    return {"count": len(PENDING_ACTIONS), "actions": PENDING_ACTIONS}
app.include_router(admin_router)
# ---------------------------------------------------------------------------# Startup Event: Seed Default Admin# ---------------------------------------------------------------------------@app.on_event("startup")def seed_default_admin():    if "admin" not in ADMIN_USERS_DB:        ADMIN_USERS_DB["admin"] = {            "username": "admin",            "hashed_password": get_password_hash("MiloAdmin2026!"),            "role": "admin"        }        print("\n" + "="*50)        print(" Default Admin seeded:")        print("Username: admin")        print("Password: MiloAdmin2026!")        print("="*50 + "\n")
# ---------------------------------------------------------------------------# Core Inbound AI Assistant Endpoint# ---------------------------------------------------------------------------@app.post("/milo/incoming")async def handle_inbound_message(data: InboundPayload):    # Intent Detection    intent_eval = openai_client.beta.chat.completions.parse(        model="gpt-4o-mini",        messages=[{"role": "system", "content": f"Classify this input: '{data.message}'"}],        response_format=TriageIntent,    ).choices[0].message.parsed
    if intent_eval.intent == "lead_qualification":        prompt = f"""        You are Milo, executive chief of staff for {data.profile.owner_name} ({data.profile.industry}).                QUALIFICATION CHECKLIST:        {chr(10).join(f"- {c}" for c in data.profile.qualifying_criteria)}                DEALBREAKERS:        {chr(10).join(f"- {d}" for d in data.profile.disqualifiers)}                CALENDAR RULES:        {data.profile.calendar_rules}                VOICE TONE:        {data.profile.brand_voice}                PROSPECT CONTACT: {data.sender_contact}        PROSPECT MESSAGE: \"\"\"{data.message}\"\"\"        """                qualification = openai_client.beta.chat.completions.parse(            model="gpt-4o-mini",            messages=[{"role": "system", "content": prompt}],            response_format=LeadQualificationResult,        ).choices[0].message.parsed
        action_id = f"lead_{int(datetime.utcnow().timestamp())}"                # Stage action        PENDING_ACTIONS[action_id] = {            "type": "lead_booking",            "contact": data.sender_contact,            "reply": qualification.suggested_reply,            "minutes_offset": qualification.proposed_minutes_from_now or 1440,            "summary": qualification.summary        }
        # Dispatch Telegram Card        await send_telegram_approval(            action_id=action_id,            title=f"Lead Fit Score: {qualification.lead_score}/100",            details=qualification.summary,            preview=qualification.suggested_reply        )
        return {"status": "staged_for_approval", "action_id": action_id, "score": qualification.lead_score}
    return {"status": "unhandled_intent", "intent": intent_eval.intent}
# ---------------------------------------------------------------------------# Telegram Webhook Endpoint# ---------------------------------------------------------------------------@app.post("/telegram/webhook")async def telegram_webhook(request: Request):    payload = await request.json()        if "callback_query" in payload:        callback = payload["callback_query"]        callback_id = callback["id"]        data_action = callback["data"]        msg = callback["message"]        chat_id = msg["chat"]["id"]        message_id = msg["message_id"]
        decision, action_id = data_action.split(":", 1)        action_data = PENDING_ACTIONS.get(action_id)
        if not action_data:            response_text = " Task expired or already resolved."        elif decision == "approve":            meeting_time = datetime.utcnow() + timedelta(minutes=action_data["minutes_offset"])            calendar_link = schedule_calendar_event(                summary=f"Discovery Call: {action_data['contact']}",                description=f"Automated booking by Milo.\nSummary: {action_data['summary']}",                start_dt=meeting_time            )            response_text = (                f" *Approved & Scheduled by Milo*\n\n"                f" Calendar Blocked: {meeting_time.strftime('%Y-%m-%d %H:%M UTC')}\n"                f" [Calendar Link]({calendar_link})\n"                f" Draft dispatched to {action_data['contact']}."            )            del PENDING_ACTIONS[action_id]        else:            response_text = " *Dismissed by operator.* No calendar action taken."            del PENDING_ACTIONS[action_id]
        if BOT_TOKEN:            async with httpx.AsyncClient() as client:                await client.post(f"{TELEGRAM_API}/answerCallbackQuery", json={"callback_query_id": callback_id})            await update_telegram_message(chat_id, message_id, response_text)
    return {"ok": True}
if __name__ == "__main__":    import uvicorn    uvicorn.run(app, host="0.0.0.0", port=8000)
